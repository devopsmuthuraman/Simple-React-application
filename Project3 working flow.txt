1. DockerFile

	# Use lightweight nginx image
	FROM nginx:alpine

	# Remove default nginx website
	  RUN rm -rf /usr/share/nginx/html/*

	# Copy project files into nginx public folder
	  COPY . /usr/share/nginx/html

	# Expose port
	  EXPOSE 80

	# Start nginx
	  CMD ["nginx", "-g", "daemon off;"]

3. Create .build.sh
	
	#!/bin/bash

	# Stop script if any command fails
	set -e

	# Variables
	IMAGE_NAME="react-app"
	IMAGE_TAG="latest"

	echo "Starting Docker build..."

	# Build Docker image
	docker build -t $IMAGE_NAME:$IMAGE_TAG .

	echo "Docker image built successfully!"
	echo "Image: $IMAGE_NAME:$IMAGE_TAG"

	# Optional: show images
	docker images | grep $IMAGE_NAME

6. Create Docker image
 
	using react project create docker image
 
	6.1.Create DockerFile
 
		#Base image
		FROM node:latest

		#Working directory
		RUN mkdir /app
		WORKDIR /app

		#Copy package to /app
		COPY package*.jason ./

		#Install npn
		RUN npm install

		#Copy all files
		COPY . .

		#Port expose
		EXPOSE 3000

		#Run the code
		CMD ["npm". "start"]
	
	6.2.Create Dockerignore file
	    
		#Dependencies and build artifacts
		 node_modules/
		 dist/
		 build

        #Version control
		 .git/
		 gitignore
		 dockerignore
		 DockerFile

		#Environment and sensitive files
		 .env
		 npm-debug.log*
		
	6.3.Build Docker image
		
		#cmd
		docker build -t react-image .
